-- Same three probes in both stages, as RCONF_CASE_APP in XEPDB1.
-- Start SQLPlus from repository root. Create local-evidence/antes first.
SET ECHO OFF
SET VERIFY OFF
SET SERVEROUTPUT ON SIZE UNLIMITED
WHENEVER OSERROR EXIT FAILURE
WHENEVER SQLERROR EXIT SQL.SQLCODE ROLLBACK
BEGIN
 IF SYS_CONTEXT('USERENV','CON_NAME') <> 'XEPDB1' THEN
  RAISE_APPLICATION_ERROR(-20040,'This case requires XEPDB1');
 END IF;
END;
/
BEGIN
 IF USER <> 'RCONF_CASE_APP' THEN RAISE_APPLICATION_ERROR(-20042,'Connect as RCONF_CASE_APP'); END IF;
END;
/
SPOOL local-evidence/antes/caso_pruebas.txt
DECLARE
 v_email VARCHAR2(120);
 PROCEDURE outcome(p_test VARCHAR2,p_result VARCHAR2,p_code NUMBER) IS
 BEGIN DBMS_OUTPUT.PUT_LINE('RCONF_CASE|TEST|BEFORE|'||p_test||'|'||p_result||'|'||p_code); END;
BEGIN
 DBMS_SESSION.SET_IDENTIFIER('rconf_case:BEFORE');
 DBMS_OUTPUT.PUT_LINE('RCONF_CASE|CONTEXT|BEFORE|'||USER||'|'||SYS_CONTEXT('USERENV','CON_NAME'));
 DBMS_OUTPUT.PUT_LINE('RCONF_CASE|TIME|BEFORE|'||TO_CHAR(SYSTIMESTAMP,'YYYY-MM-DD"T"HH24:MI:SSTZH:TZM'));
 DBMS_OUTPUT.PUT_LINE('RCONF_CASE|RUN|BEFORE|'||RAWTOHEX(SYS_GUID()));
 BEGIN
  EXECUTE IMMEDIATE 'SELECT email FROM rconf_case_owner.case_clients WHERE id=1' INTO v_email;
  outcome('RAW_SELECT','ALLOWED',0);
  DBMS_OUTPUT.PUT_LINE('SYNTHETIC_EMAIL|'||v_email);
 EXCEPTION WHEN OTHERS THEN outcome('RAW_SELECT','DENIED',ABS(SQLCODE)); END;
 BEGIN
  EXECUTE IMMEDIATE 'SELECT email FROM rconf_case_owner.case_masked WHERE id=1' INTO v_email;
  IF v_email='[oculto]' THEN outcome('MASKED_SELECT','MASKED',0);
  ELSE outcome('MASKED_SELECT','UNEXPECTED',0); END IF;
 EXCEPTION WHEN OTHERS THEN outcome('MASKED_SELECT','DENIED',ABS(SQLCODE)); END;
 BEGIN
  EXECUTE IMMEDIATE 'UPDATE rconf_case_owner.case_clients SET email=email WHERE 1=0';
  outcome('DIRECT_UPDATE','ALLOWED',0);
 EXCEPTION WHEN OTHERS THEN outcome('DIRECT_UPDATE','DENIED',ABS(SQLCODE)); END;
 ROLLBACK;
 DBMS_SESSION.CLEAR_IDENTIFIER;
 DBMS_OUTPUT.PUT_LINE('RCONF_CASE|END|BEFORE|3');
END;
/
SPOOL OFF
